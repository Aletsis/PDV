@code {
    private ClientDto? _selectedClient;
    private string _paymentMethod = "Cash";
    private List<CartItem> _cart = new();

    private List<Order> _openOrders = new();

    private string searchCode = "";
    private ProductDto? searchedProduct;

    private decimal weightToAdd = 1.0m;
    private bool readingScale = false;

    private List<ClientDto> _allClients = new();
    private List<ProductDto> _allProducts = new();
    private List<DeliveryZone> _zones = new();

    private decimal _subtotal => _cart.Sum(x => x.Total);
    private decimal _taxes => _cart.Sum(x => x.Total * GetTaxRatePercent(x.Product.TaxRate) / 100m);
    private decimal _deliveryCost = 0m;
    private decimal _total => _subtotal + _taxes + _deliveryCost;

    private bool CanSubmit => _selectedClient != null && _cart.Count > 0 && ShiftSession.HasActiveShift;

    protected override async Task OnInitializedAsync()
    {
        if (!ShiftSession.HasActiveShift)
        {
            var authState = await AuthStateProvider.GetAuthenticationStateAsync();
            var user = authState.User;
            var currentUserId = user.FindFirst(System.Security.Claims.ClaimTypes.NameIdentifier)?.Value;

            ShiftDto? activeShift = null;

            // 1. Intentar restaurar por ID del usuario autenticado
            if (!string.IsNullOrEmpty(currentUserId))
            {
                activeShift = await Mediator.Send(new GetActiveShiftByUserIdQuery(currentUserId));
            }

            // 2. Si no se encontró por usuario, intentar restaurar por IP de la terminal
            if (activeShift == null)
            {
                try
                {
                    var httpContext = HttpContextAccessor.HttpContext;
                    var rawIp = httpContext?.Connection.RemoteIpAddress?.MapToIPv4().ToString();
                    var clientIp = IpAddressHelper.ResolveClientIp(rawIp);
                    if (!string.IsNullOrEmpty(clientIp))
                    {
                        var cashRegister = await Mediator.Send(new GetCashRegisterByIpQuery(clientIp));
                        if (cashRegister != null)
                        {
                            activeShift = await Mediator.Send(new GetActiveShiftByCashRegisterQuery(cashRegister.Id));
                        }
                    }
                }
                catch
                {
                    // Fallback silencioso
                }
            }

            // Si se encontró turno activo, hidratar ShiftSession
            if (activeShift != null)
            {
                ShiftSession.SetActive(
                    activeShift.Id,
                    activeShift.CashRegisterId,
                    activeShift.CashRegisterName,
                    activeShift.InitialCash,
                    activeShift.StartTime,
                    activeShift.BranchId);
            }
        }

        if (ShiftSession.HasActiveShift)
        {
            await LoadDataAsync();
        }
    }

    private async Task LoadDataAsync()
    {
        _allClients = await Mediator.Send(new ListClientsQuery { IncludeInactive = false });
        _allProducts = await Mediator.Send(new GetProductsQuery());
        _zones = await Mediator.Send(new GetDeliveryZonesQuery { OnlyActive = true });
        await RefreshOpenOrders();
    }

    private Task<IEnumerable<ClientDto>> SearchClients(string value, CancellationToken token)
    {
        if (string.IsNullOrEmpty(value))
            return Task.FromResult<IEnumerable<ClientDto>>(_allClients);

        var search = value.Trim().ToLower();
        return Task.FromResult<IEnumerable<ClientDto>>(
            _allClients.Where(x => x.Name.ToLower().Contains(search) || x.Code.ToLower().Contains(search))
        );
    }

    private void OnClientChanged(ClientDto client)
    {
        _selectedClient = client;
        if (client != null && client.DeliveryZoneId.HasValue)
        {
            var zone = _zones.FirstOrDefault(z => z.Id == client.DeliveryZoneId.Value);
            _deliveryCost = zone?.DeliveryCost ?? 0m;
        }
        else
        {
            _deliveryCost = 0m;
        }

        // Al cambiar de cliente, volvemos a calcular los precios de los productos en el carrito (por si cambió a/de Wholesale)
        foreach (var item in _cart)
        {
            decimal? priceOverride = null;
            if (_selectedClient != null)
            {
                if (_selectedClient.ClientType == ClientType.Wholesale && item.Product.WholesalePrice.HasValue)
                {
                    priceOverride = item.Product.WholesalePrice.Value;
                }
            }
            item.PriceOverride = priceOverride;
        }
        StateHasChanged();
    }

    private async Task KeyPress(KeyboardEventArgs e)
    {
        if (e.Key == "Enter")
        {
            await SearchProduct();
        }
        else if (e.Key == "F9")
        {
            await TriggerScaleRead();
        }
        else if (e.Key == "F3")
        {
            await OpenProductLookup();
        }
    }

    public async Task TriggerScaleRead()
    {
        if (searchedProduct != null && searchedProduct.SaleType == "Bulk")
        {
            await ReadFromScale();
        }
    }

    private async Task ReadFromScale()
    {
        if (readingScale) return;
        readingScale = true;
        StateHasChanged();
        try
        {
            var port = Configuration["Hardware:Scale:Port"] ?? "MOCK";
            var baud = int.TryParse(Configuration["Hardware:Scale:BaudRate"], out var b) ? b : 9600;
            var protocol = Configuration["Hardware:Scale:Protocol"] ?? "Mock";

            var result = await ScaleService.ReadWeightAsync(port, baud, protocol);
            if (result.Success)
            {
                weightToAdd = result.Weight;
                Snackbar.Add($"Peso leído: {result.Weight} {result.Unit}", Severity.Success);
            }
            else
            {
                Snackbar.Add($"Error de báscula: {result.ErrorMessage}", Severity.Error);
            }
        }
        catch (Exception ex)
        {
            Snackbar.Add($"Error al leer báscula: {ex.Message}", Severity.Error);
        }
        finally
        {
            readingScale = false;
            StateHasChanged();
        }
    }

    private async Task SearchProduct()
    {
        if (string.IsNullOrWhiteSpace(searchCode)) return;

        notFound = false;
        searchedProduct = null;
        var processedCode = searchCode.Trim();
        searchCode = ""; // Limpiar inmediatamente para escaneo continuo fluido
        StateHasChanged();

        // EAN-13 Bulk Barcode Parsing: starts with '2', has 13 numeric digits
        if (processedCode.Length == 13 && processedCode.StartsWith("2") && processedCode.All(char.IsDigit))
        {
            var codePart = processedCode.Substring(1, 6);
            var weightPart = processedCode.Substring(7, 6);

            if (int.TryParse(weightPart.Substring(0, 2), out int integerPart) && 
                int.TryParse(weightPart.Substring(2, 4), out int fractionalPart))
            {
                decimal weight = integerPart + (decimal)fractionalPart / 10000m;

                var product = await Mediator.Send(new GetProductByCodeQuery(codePart));
                if (product != null)
                {
                    var productDto = MapToDto(product);
                    AddProductDirectly(productDto, weight);
                    if (barcodeField != null)
                    {
                        await barcodeField.FocusAsync();
                    }
                    return;
                }
            }
        }

        // Standard flow
        var foundProduct = await Mediator.Send(new GetProductByCodeQuery(processedCode));

        if (foundProduct == null)
        {
            notFound = true;
            if (barcodeField != null)
            {
                await barcodeField.FocusAsync();
            }
        }
        else
        {
            var productDto = MapToDto(foundProduct);
            // Auto add Piece products to cart for ultimate speed!
            if (productDto.SaleType == "Piece")
            {
                AddProductDirectly(productDto, 1);
                if (barcodeField != null)
                {
                    await barcodeField.FocusAsync();
                }
            }
            else
            {
                searchedProduct = productDto;
                weightToAdd = 1.0m;
                searchedProductStock = 0;
                if (ShiftSession.BranchId.HasValue)
                {
                    var branchStock = await DbContext.ProductBranchStocks
                        .FirstOrDefaultAsync(s => s.ProductId == foundProduct.Id && s.BranchId == ShiftSession.BranchId.Value);
                    if (branchStock != null)
                    {
                        searchedProductStock = branchStock.Stock;
                    }
                }
            }
        }
    }

    private ProductDto MapToDto(Product x)
    {
        var existing = _allProducts.FirstOrDefault(p => p.Id == x.Id);
        if (existing != null) return existing;

        return new ProductDto
        {
            Id = x.Id,
            Name = x.Name,
            Code = x.Code,
            Plu = x.Plu,
            Description = x.Description,
            Price = x.Price,
            WholesalePrice = x.WholesalePrice,
            WholesaleMinQuantity = x.WholesaleMinQuantity,
            SaleType = x.SaleType.ToString(),
            Barcode = x.Barcode,
            Cost = x.Cost,
            TaxRate = x.TaxRate.ToString(),
            IsActive = x.IsActive,
            SatCode = x.SatCode,
            Type = (int)x.Type,
            ControlExistencia = (int)x.ControlExistencia,
            SaleUnitId = x.SaleUnitId,
            SaleUnitName = x.SaleUnitName,
            XmlUnitId = x.XmlUnitId,
            Department = x.Department,
            Clasificacion1Id = x.Clasificacion1Id,
            Clasificacion5Id = x.Clasificacion5Id
        };
    }

    private void AddProductDirectly(ProductDto product, decimal quantityOrWeight)
    {
        decimal? priceOverride = null;
        if (_selectedClient != null)
        {
            if (_selectedClient.ClientType == ClientType.Wholesale && product.WholesalePrice.HasValue)
            {
                priceOverride = product.WholesalePrice.Value;
            }
        }

        var existingItem = _cart.FirstOrDefault(x => x.Product.Id == product.Id);
        if (existingItem != null)
        {
            existingItem.Quantity += quantityOrWeight;
            Snackbar.Add($"{product.Name} actualizado a {existingItem.Quantity.ToString(product.SaleType == "Bulk" ? "N3" : "N0")}", Severity.Success);
        }
        else
        {
            _cart.Add(new CartItem
            {
                Product = product,
                Quantity = quantityOrWeight,
                PriceOverride = priceOverride
            });
            Snackbar.Add($"{product.Name} agregado al pedido", Severity.Success);
        }
        StateHasChanged();
    }

    private void AddToCart()
    {
        if (searchedProduct == null) return;
        var prod = searchedProduct;
        decimal amount = prod.SaleType == "Bulk" ? weightToAdd : quantityToAdd;

        searchedProduct = null;
        searchCode = "";
        notFound = false;
        quantityToAdd = 1;
        weightToAdd = 1.0m;

        AddProductDirectly(prod, amount);

        if (barcodeField != null)
        {
            InvokeAsync(async () => {
                await Task.Delay(50);
                await barcodeField.FocusAsync();
            });
        }
    }

    private async Task RemoveFromCart(CartItem item)
    {
        var auth = await RequestAuthorizationAsync($"Eliminar producto '{item.Product.Name}' del pedido", "orders.remove_item");
        if (auth.Success)
        {
            _cart.Remove(item);
            Snackbar.Add($"{item.Product.Name} eliminado del pedido", Severity.Warning);
        }
    }

    private void ResetOrderState()
    {
        _cart.Clear();
        _selectedClient = null;
        searchedProduct = null;
        searchCode = "";
        notFound = false;
        quantityToAdd = 1;
        weightToAdd = 1.0m;
        _deliveryCost = 0m;
        _paymentMethod = "Cash";
        _activeOrderId = null;
    }

    private async Task<bool> SubmitOrderInternal(bool printTicket)
    {
        if (!CanSubmit) return false;

        try
        {
            var authState = await AuthStateProvider.GetAuthenticationStateAsync();
            var user = authState.User;
            string userId = user.FindFirst(System.Security.Claims.ClaimTypes.NameIdentifier)?.Value ?? user.Identity?.Name ?? "system";

            if (_activeOrderId.HasValue)
            {
                // Cancelar la versión anterior antes de registrar la nueva versión editada
                var cancelResult = await Mediator.Send(new CancelOrderCommand 
                { 
                    OrderId = _activeOrderId.Value, 
                    Reason = "Modificación de Pedido" 
                });
                
                if (!cancelResult)
                {
                    Snackbar.Add("No se pudo actualizar el pedido porque falló la cancelación de la versión anterior.", Severity.Error);
                    return false;
                }
            }

            var command = new CreateOrderCommand
            {
                ClientId = _selectedClient!.Id,
                CashRegisterId = ShiftSession.ActiveCashRegisterId ?? Guid.Empty,
                PaymentMethod = _paymentMethod,
                UserId = userId,
                Items = _cart.Select(x => new OrderCartItemDto
                {
                    ProductId = x.Product.Id,
                    Quantity = x.Quantity,
                    PriceOverride = x.PriceOverride
                }).ToList()
            };

            var orderId = await Mediator.Send(command);

            if (printTicket)
            {
                Snackbar.Add(_activeOrderId.HasValue ? "Pedido actualizado exitosamente." : "Pedido registrado exitosamente.", Severity.Success);
                // Imprimir el ticket de forma asíncrona
                await Mediator.Send(new PrintOrderTicketCommand(orderId, ShiftSession.ActiveCashRegisterId ?? Guid.Empty));
            }

            return true;
        }
        catch (Exception ex)
        {
            Snackbar.Add($"Error al registrar/actualizar pedido: {ex.Message}", Severity.Error);
            return false;
        }
    }

    private async Task SubmitOrder()
    {
        var success = await SubmitOrderInternal(printTicket: true);
        if (success)
        {
            ResetOrderState();
            await RefreshOpenOrders();
        }
    }

    private async Task NuevoPedido()
    {
        if (_cart.Count == 0) return;

        if (_activeOrderId == null)
        {
            if (_selectedClient == null)
            {
                Snackbar.Add("Debe seleccionar un cliente para guardar el pedido.", Severity.Warning);
                return;
            }

            // Registrar pedido en la BD (como confirmado/abierto) sin imprimir ticket
            var success = await SubmitOrderInternal(printTicket: false);
            if (success)
            {
                ResetOrderState();
                Snackbar.Add("Pedido guardado en pedidos abiertos", Severity.Info);
                await RefreshOpenOrders();
            }
        }
        else
        {
            // Ya estaba cargado un pedido existente, simplemente limpiamos la pantalla
            ResetOrderState();
            Snackbar.Add("Pantalla limpiada. El pedido sigue abierto.", Severity.Info);
        }
    }

    private async Task CancelarPedido()
    {
        if (_cart.Count == 0) return;

        if (_activeOrderId.HasValue)
        {
            var auth = await RequestAuthorizationAsync($"Cancelar pedido registrado", "orders.cancel_order");
            if (auth.Success)
            {
                try
                {
                    var result = await Mediator.Send(new CancelOrderCommand { OrderId = _activeOrderId.Value, Reason = "Cancelado desde captura" });
                    if (result)
                    {
                        ResetOrderState();
                        Snackbar.Add("Pedido cancelado exitosamente.", Severity.Error);
                        await RefreshOpenOrders();
                    }
                }
                catch (Exception ex)
                {
                    Snackbar.Add($"Error al cancelar pedido: {ex.Message}", Severity.Error);
                }
            }
        }
        else
        {
            var auth = await RequestAuthorizationAsync("Limpiar pedido completo", "orders.clear_order");
            if (auth.Success)
            {
                ResetOrderState();
                Snackbar.Add("Pedido limpiado", Severity.Info);
            }
        }
    }

    private async Task ResumeOrder(Order order)
    {
        // Cargar pedido con sus artículos de la BD
        var orderWithItems = await DbContext.Orders
            .Include(o => o.Items)
            .ThenInclude(i => i.Product)
            .FirstOrDefaultAsync(o => o.Id == order.Id);

        if (orderWithItems == null)
        {
            Snackbar.Add("No se pudo cargar el pedido.", Severity.Error);
            return;
        }

        _activeOrderId = orderWithItems.Id;
        _selectedClient = _allClients.FirstOrDefault(c => c.Id == orderWithItems.ClientId);
        _paymentMethod = orderWithItems.PaymentMethod.ToString();
        
        if (_selectedClient != null && _selectedClient.DeliveryZoneId.HasValue)
        {
            var zone = _zones.FirstOrDefault(z => z.Id == _selectedClient.DeliveryZoneId.Value);
            _deliveryCost = zone?.DeliveryCost ?? 0m;
        }
        else
        {
            _deliveryCost = 0m;
        }

        // Cargar carrito
        _cart.Clear();
        foreach (var item in orderWithItems.Items)
        {
            if (item.Product != null)
            {
                var productDto = MapToDto(item.Product);
                _cart.Add(new CartItem
                {
                    Product = productDto,
                    Quantity = item.Quantity,
                    PriceOverride = item.UnitPrice != productDto.Price ? item.UnitPrice : null
                });
            }
        }

        Snackbar.Add($"Pedido #{orderWithItems.Series}-{orderWithItems.Folio} cargado", Severity.Success);
        StateHasChanged();
    }

    private async Task RefreshOpenOrders()
    {
        if (ShiftSession.HasActiveShift)
        {
            var branchId = ShiftSession.BranchId ?? Guid.Empty;
            _openOrders = await Mediator.Send(new GetOrdersQuery 
            { 
                Status = OrderStatus.Confirmed,
                BranchId = branchId
            });
        }
        else
        {
            _openOrders.Clear();
        }
    }

    private async Task OpenProductLookup()
    {
        var parameters = new DialogParameters<ProductLookupDialog>();
        if (ShiftSession.BranchId.HasValue)
        {
            parameters.Add(x => x.BranchId, ShiftSession.BranchId.Value);
        }
        var dialog = await DialogService.ShowAsync<ProductLookupDialog>("Buscar Producto", parameters, new DialogOptions { MaxWidth = MaxWidth.Medium, FullWidth = true, CloseOnEscapeKey = true });
        var result = await dialog.Result;
        if (result != null && !result.Canceled && result.Data is ProductLookupDialog.ProductLookupResult lookupResult)
        {
            var product = lookupResult.Product;
            var qty = lookupResult.QuantityOrWeight;
            var productDto = MapToDto(product);
            AddProductDirectly(productDto, qty);
        }
    }

    private async Task<AuthorizationDialog.AuthorizationResult> RequestAuthorizationAsync(string actionName, string requiredPermissionCode)
    {
        var authState = await AuthStateProvider.GetAuthenticationStateAsync();
        var user = authState.User;
        if (user.Identity?.IsAuthenticated == true && (user.IsInRole("Admin") || user.IsInRole("Manager")))
        {
            return new AuthorizationDialog.AuthorizationResult(true, null, null);
        }

        var parameters = new DialogParameters { ["ActionName"] = actionName };
        var options = new DialogOptions { MaxWidth = MaxWidth.Small, FullWidth = true, CloseOnEscapeKey = false, BackdropClick = false };
        var dialog = await DialogService.ShowAsync<PDV.WebUI.Components.Pages.Auth.AuthorizationDialog>("Autorización Requerida", parameters, options);
        var result = await dialog.Result;
        
        if (result != null && !result.Canceled && result.Data is AuthorizationDialog.AuthorizationResult authResult && authResult.Success)
        {
            return authResult;
        }

        return new AuthorizationDialog.AuthorizationResult(false, null, null);
    }

    private decimal GetTaxRatePercent(string taxRate)
    {
        return taxRate switch
        {
            "Rate16" => 16m,
            "Rate8" => 8m,
            _ => 0m
        };
    }

    private class CartItem
    {
        public ProductDto Product { get; set; } = null!;
        public decimal Quantity { get; set; }
        public decimal? PriceOverride { get; set; }
        public decimal Total => Quantity * (PriceOverride ?? Product.Price);
    }
